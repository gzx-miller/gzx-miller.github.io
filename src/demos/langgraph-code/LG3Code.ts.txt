// 中间件：用钩子函数扩展智能体工作流
// 场景：给课程咨询智能体加上四个中间件——预算控制、消息截断、敏感词过滤、PII 检测
import { createAgent, createMiddleware } from 'langchain'
import { ChatOpenAI } from '@langchain/openai'
import { tool } from '@langchain/core/tools'
import { HumanMessage, AIMessage, SystemMessage } from '@langchain/core/messages'
import { z } from 'zod'

const model = new ChatOpenAI({ model: 'gpt-4o-mini', temperature: 0 })

const searchCourses = tool(
  async ({ keyword }) => `与「${keyword}」相关的课程：Vue3 入门、Vue3 组件通信`,
  {
    name: 'search_courses',
    description: '按关键词搜索课程库',
    schema: z.object({ keyword: z.string() }),
  },
)

// 示例1: 预算控制——beforeModel 统计累计 token，超预算直接熔断
const budgetGuard = createMiddleware({
  name: 'BudgetGuard',
  beforeModel: (state) => {
    const spent = state.messages.reduce((total: number, m: any) => total + (m.usage_metadata?.total_tokens ?? 0), 0)
    if (spent > 20_000) {
      // 返回 jumpTo 跳转到终止节点，不再调用模型
      return { jumpTo: 'end' }
    }
    return
  },
})

// 示例2: 消息截断——wrapModelCall 里只保留最近 N 条消息，防止上下文撑爆
const trimMessages = createMiddleware({
  name: 'TrimMessages',
  wrapModelCall: async (request, handler) => {
    const MAX_MESSAGES = 12
    const trimmed =
      request.messages.length > MAX_MESSAGES
        ? [request.messages[0], ...request.messages.slice(-MAX_MESSAGES + 1)]
        : request.messages
    const request2 = { ...request, messages: trimmed }
    return handler(request2)
  },
})

// 示例3: 敏感词过滤——afterModel 检查模型输出，命中敏感词就替换成安全话术
const SENSITIVE_WORDS = ['保证提分', '全网最低价', '绝对通过']
const contentFilter = createMiddleware({
  name: 'ContentFilter',
  afterModel: (state) => {
    const last = state.messages.at(-1) as AIMessage | undefined
    if (!last || typeof last.content !== 'string') return
    const hit = SENSITIVE_WORDS.find((word) => last.content.includes(word))
    if (hit) {
      return {
        messages: [new AIMessage('抱歉，该表述不符合规范，已自动替换。')],
      }
    }
    return
  },
})

// 示例4: PII 检测——wrapModelCall 扫描输入，手机号、邮箱先脱敏再送给模型
const piiGuard = createMiddleware({
  name: 'PiiGuard',
  wrapModelCall: async (request, handler) => {
    const masked = request.messages.map((m: any) => {
      if (typeof m.content !== 'string') return m
      const content = m.content
        .replace(/(\d{3})\d{4}(\d{4})/g, '$1****$2')
        .replace(/[\w.-]+@[\w.-]+\.\w+/g, '***@***.***')
      return new m.constructor({ ...m, content })
    })
    return handler({ ...request, messages: masked })
  },
})

// 示例5: 把四个中间件挂到智能体上，按数组顺序依次生效
const agent = createAgent({
  model,
  tools: [searchCourses],
  middleware: [budgetGuard, trimMessages, contentFilter, piiGuard],
  systemPrompt: '你是一名课程咨询助手，回答前先检索课程库。',
})

// 示例6: 调用验证——手机号被脱敏、超预算被熔断、敏感表述被替换
const result = await agent.invoke({
  messages: [new HumanMessage('我的手机号是 13812345678，帮我推荐 Vue3 课程')],
})
console.log(result.messages.at(-1)?.content)
