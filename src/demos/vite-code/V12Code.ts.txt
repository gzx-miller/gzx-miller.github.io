// vite.config.ts - CSS 相关配置
import { defineConfig } from 'vite'
import vue from '@vitejs/plugin-vue'
import autoprefixer from 'autoprefixer'
import postcssNested from 'postcss-nested'
import tailwindcss from '@tailwindcss/postcss'
// ESM 配置文件统一用 import，不再使用 require()

export default defineConfig({
  plugins: [vue()],
  css: {
    // PostCSS 配置（也可以用 postcss.config.mjs）
    postcss: {
      plugins: [
        // Tailwind v4 的 PostCSS 适配器
        tailwindcss(),
        // 自动添加浏览器前缀
        autoprefixer(),
        // CSS 嵌套
        postcssNested()
      ]
    },
    
    // CSS Modules 配置
    modules: {
      // 生成的类名格式
      generateScopedName: '[name]__[local]___[hash:base64:5]',
      // 是否使用 camelCase
      localsConvention: 'camelCase'
    },
    
    // 预处理器配置
    preprocessorOptions: {
      scss: {
        // 全局注入的变量和 mixin；Vite 7 只保留 Sass modern API，
        // 因此用 @use 替代已废弃的 @import
        additionalData: `
          @use "@/styles/variables.scss" as *;
          @use "@/styles/mixins.scss" as *;
        `
        // Vite 7 已移除 Sass legacy API，api: 'modern-compiler' 选项不再需要
      },
      less: {
        // Less 全局变量
        modifyVars: {
          'primary-color': '#1890ff'
        },
        javascriptEnabled: true
      }
    }
  }
})

// ====================
// PostCSS 配置文件 (postcss.config.mjs)
// ====================

export default {
  plugins: {
    '@tailwindcss/postcss': {},
    autoprefixer: {}
  }
}
