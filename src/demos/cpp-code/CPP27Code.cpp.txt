#include <iostream>

constexpr int factorial(int n) {
    return (n <= 1) ? 1 : n * factorial(n - 1);
}

consteval int alwaysCompileTime(int x) {
    return x * 2;
}

int main() {
    constexpr int result = factorial(10);
    std::cout << "10! = " << result << std::endl;
    
    int x = 21;
    int y = alwaysCompileTime(x);
    std::cout << "alwaysCompileTime(21) = " << y << std::endl;
    
    return 0;
}
