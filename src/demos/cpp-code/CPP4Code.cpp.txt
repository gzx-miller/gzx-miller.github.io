// ═══════════════════════════════════════════
// 1. if / else if / else 互斥多路选择
// ═══════════════════════════════════════════

#include <iostream>

int main() {
    int score = 95;
    if (score >= 90) {                      // 从上到下，第一个为真的分支胜出
        std::cout << "优秀";
    } else if (score >= 60) {               // 仅在前一条件为假时才判断
        std::cout << "及格";
    } else {
        std::cout << "不及格";
    }                                       // 结果：优秀
    return 0;
}

// ═══════════════════════════════════════════
// 2. switch 与 case 穿透
// ═══════════════════════════════════════════

int main() {
    int day = 3;
    switch (day) {                          // 只对整型常量做等值匹配
        case 1: std::cout << "周一"; break; // 结果：打印后跳出 switch
        case 2: std::cout << "周二"; break;
        case 3: std::cout << "周三"; break; // 结果：命中并打印「周三」
        default: std::cout << "其他"; break;
    }

    int level = 1;
    switch (level) {
        case 1:                             // 故意不写 break：1 与 2 共用逻辑
        case 2: std::cout << "低段位"; break;  // 结果：1 也打印「低段位」
        default: break;
    }
    return 0;
}

// ═══════════════════════════════════════════
// 3. while 与 do-while
// ═══════════════════════════════════════════

int main() {
    int i = 0;
    while (i < 5) {                         // 先判后做，可能一次都不执行
        std::cout << i << " ";              // 结果：0 1 2 3 4
        ++i;
    }

    int j = 10;
    do {                                    // 先做后判，循环体至少执行一次
        std::cout << j << " ";              // 结果：10
        ++j;
    } while (j < 5);                        // 条件为假，只跑了一轮
    return 0;
}

// ═══════════════════════════════════════════
// 4. for 循环与 break / continue
// ═══════════════════════════════════════════

int main() {
    for (int i = 0; i < 5; ++i) {           // 初始化、条件、步进集中在头部
        if (i == 2) continue;               // 跳过本轮剩余语句
        if (i == 4) break;                  // 立即跳出整个循环
        std::cout << i << " ";              // 结果：0 1 3
    }

    int n = 0, sum = 0;
    while (n < 100) {                       // 条件驱动
        sum += n;
        ++n;
    }                                       // 结果：sum 为 0..99 之和 4950
    return 0;
}

// ═══════════════════════════════════════════
// 5. 范围 for 与引用修改
// ═══════════════════════════════════════════

#include <vector>

int main() {
    std::vector<int> nums{1, 2, 3, 4, 5};

    for (int num : nums) {                  // 按值拷贝，修改不影响容器
        num *= 10;                          // 结果：nums 仍为 1 2 3 4 5
    }

    for (const auto& num : nums) {          // 只读引用，避免拷贝
        std::cout << num << " ";            // 结果：1 2 3 4 5
    }

    for (auto& num : nums) {                // 可写引用，改的是容器里的元素
        num *= 2;                           // 结果：nums 变为 2 4 6 8 10
    }
    return 0;
}

// ═══════════════════════════════════════════
// 6. C++17 在 if / switch 中初始化限制作用域
// ═══════════════════════════════════════════

#include <map>
#include <string>

int main() {
    std::map<std::string, int> stock{{"apple", 3}};
    if (auto it = stock.find("apple"); it != stock.end()) {  // 初始化写在条件里
        std::cout << it->second;            // 结果：3，it 作用域限于此 if
    }                                       // 出 if 后 it 不再可见
    return 0;
}
