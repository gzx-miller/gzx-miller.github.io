// ═══════════════════════════════════════════
// 1. 取址与解引用
// ═══════════════════════════════════════════

#include <iostream>

int main() {
    int x = 42;
    int* p = &x;                            // 取址：p 保存 x 的地址
    std::cout << *p << std::endl;           // 结果：42，解引用读到 x 的值
    *p = 100;                               // 通过指针写回原对象
    std::cout << x << std::endl;            // 结果：100，x 已被修改
    return 0;
}

// ═══════════════════════════════════════════
// 2. 空指针 nullptr
// ═══════════════════════════════════════════

int main() {
    int* np = nullptr;                      // C++11 起的有类型空指针
    if (np != nullptr) {                    // 解引用前先判空
        std::cout << *np << std::endl;      // 结果：条件为假，这句不会执行
    }
    np = nullptr;                           // 对象不存在时把指针置空
    return 0;
}

// ═══════════════════════════════════════════
// 3. 指针算术与数组关系
// ═══════════════════════════════════════════

int main() {
    int arr[] = {10, 20, 30};
    int* q = arr;                           // 数组名退化为指向首元素的指针
    std::cout << *q << std::endl;           // 结果：10
    std::cout << *(q + 1) << std::endl;     // 结果：20，前进一个元素而非一个字节
    std::cout << arr[1] << std::endl;       // 结果：20，等价于 *(arr + 1)
    q += 2;                                 // 指针后移两个元素
    std::cout << *q << std::endl;           // 结果：30
    return 0;
}

// ═══════════════════════════════════════════
// 4. const 指针 vs 指向 const 的指针
// ═══════════════════════════════════════════

int main() {
    int a = 1, b = 2;
    const int* p1 = &a;                     // 指向 const int：不能经 p1 改值
    // *p1 = 5;                             // 编译错误：目标被 const 约束
    p1 = &b;                                // 结果：合法，指针本身可以重指向

    int* const p2 = &a;                     // const 指针：p2 不能再指向别处
    // p2 = &b;                             // 编译错误：指针本身被 const 约束
    *p2 = 5;                                // 结果：合法，可通过 p2 修改目标

    const int* const p3 = &a;               // 两者兼有：既不能重指向也不能改值
    return 0;
}

// ═══════════════════════════════════════════
// 5. 引用即别名
// ═══════════════════════════════════════════

#include <string>

int main() {
    int y = 5;
    int& r = y;                             // r 是 y 的别名，必须初始化
    r = 10;                                 // 修改 r 即修改 y
    std::cout << y << std::endl;            // 结果：10

    std::string s = "hello";
    for (char& ch : s) {                    // 引用让遍历可直接改元素
        ch = static_cast<char>(ch - 32);    // 小写转大写
    }                                       // 结果：s 变为 HELLO
    return 0;
}
