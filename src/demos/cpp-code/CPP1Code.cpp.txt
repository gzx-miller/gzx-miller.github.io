// ═══════════════════════════════════════════
// 1. 最小程序：入口 main 与标准输出
// ═══════════════════════════════════════════

#include <iostream>                            // 引入输入输出库，供 std::cout 使用

int main() {                                   // 程序入口，操作系统从这里开始调用
    std::cout << "Hello, C++!" << std::endl;   // 输出：Hello, C++
    return 0;                                  // 退出码 0，表示正常结束
}

// ═══════════════════════════════════════════
// 2. 预处理指令：只做文本替换
// ═══════════════════════════════════════════

#define PI 3.14159                             // 宏定义，预处理阶段原地文本替换

#include <cmath>                               // 再引入一个标准库头文件

// ═══════════════════════════════════════════
// 3. 先声明、后使用：函数原型与定义
// ═══════════════════════════════════════════

int add(int a, int b);                         // 声明：只告诉编译器这个函数存在

int add(int a, int b) {                        // 定义：给出具体实现
    return a + b;                              // add(2, 3) 得到 5
}

// ═══════════════════════════════════════════
// 4. 变量、常量与块作用域
// ═══════════════════════════════════════════

int globalCount = 0;                           // 全局变量，生命周期贯穿整个程序

int main() {
    const int radius = 3;                      // 常量，初始化后不可再修改
    int area = static_cast<int>(PI * radius * radius);   // 3.14159 × 9 → 28
    {
        int area = 0;                          // 内层块作用域，遮蔽外层同名变量
        area = 1;                              // 只影响内层，函数退出即销毁
    }
    return 0;
}

// ═══════════════════════════════════════════
// 5. 编译四阶段与常用命令
// ═══════════════════════════════════════════

// g++ -c main.s -o main.o       汇编：翻译成机器码，得到目标文件
