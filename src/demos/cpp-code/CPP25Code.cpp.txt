#include <iostream>
#include <thread>
#include <mutex>
#include <vector>
#include <future>

int main() {
    // 创建线程
    std::thread t1([]() {
        std::cout << "线程 1 正在执行" << std::endl;
    });
    t1.join();

    // 使用 async 异步任务
    std::future<int> result = std::async([]() {
        return 42;
    });
    std::cout << "结果：" << result.get() << std::endl;

    return 0;
}
