// ========== 基础用法（basicCode） ==========
// @content 基本用法 — 向 Mixin 注入内容块
@mixin card {
  padding: 16px;
  border-radius: 12px;
  background: #fff;
  box-shadow: 0 2px 8px rgba(0,0,0,.08);
  
  @content; // 内容块插入位置
}

// 使用 — 在 @include 的 {} 中写额外样式
.product-card {
  @include card {
    border: 2px solid #e85d04;
    .price { color: #e85d04; font-size: 20px; }
  }
}

.user-card {
  @include card {
    display: flex;
    gap: 12px;
    .avatar { width: 48px; height: 48px; border-radius: 50%; }
  }
}

// 编译结果
.product-card {
  padding: 16px; border-radius: 12px; background: #fff; ...
  border: 2px solid #e85d04;
}
.product-card .price { color: #e85d04; font-size: 20px; }

.user-card {
  padding: 16px; border-radius: 12px; background: #fff; ...
  display: flex; gap: 12px;
}

// ========== 媒体查询（mediaCode） ==========
// 经典应用：媒体查询 Mixin
@mixin respond-to($breakpoint) {
  @if $breakpoint == mobile {
    @media (max-width: 480px) { @content; }
  } @else if $breakpoint == tablet {
    @media (max-width: 768px) { @content; }
  } @else if $breakpoint == desktop {
    @media (min-width: 1024px) { @content; }
  }
}

// 使用
.sidebar {
  width: 300px;
  
  @include respond-to(tablet) {
    width: 100%;
    position: fixed;
    bottom: 0;
  }
  
  @include respond-to(mobile) {
    display: none;
  }
}

// 编译结果
.sidebar { width: 300px; }

@media (max-width: 768px) {
  .sidebar { width: 100%; position: fixed; bottom: 0; }
}

@media (max-width: 480px) {
  .sidebar { display: none; }
}

// ========== 高级用法（advancedCode） ==========
// 高级用法：内容块 + 参数传递（Sass 3.5+）
@mixin theme($mode) {
  @if $mode == dark {
    background: #1a1a2e;
    color: #e0e0e0;
    @content (dark);
  } @else {
    background: #fff;
    color: #333;
    @content (light);
  }
}

// 使用时接收参数
.card {
  @include theme(dark) using ($mode) {
    border-color: if($mode == dark, #333, #ddd);
    .title { color: if($mode == dark, #f4a261, #e85d04); }
  }
}

// 另一个例子：hover wrapper
@mixin hover-effect {
  transition: transform 0.2s, box-shadow 0.2s;
  
  &:hover {
    transform: translateY(-2px);
    box-shadow: 0 8px 24px rgba(0,0,0,.15);
    @content;
  }
}

.btn {
  @include hover-effect {
    background: darken(#e85d04, 10%);
  }
}
