// ========== @use 导入（useCode） ==========
// @use — 现代模块导入（推荐）

// _variables.scss
$primary: #e85d04;
$secondary: #f4a261;
$radius: 8px;

// 使用 @use 导入 — 默认带命名空间
@use 'variables';

.btn {
  background: variables.$primary;
  border-radius: variables.$radius;
}

// 重命名命名空间
@use 'variables' as vars;

.btn {
  color: vars.$secondary;
}

// 导入到全局（不推荐，但实用）
@use 'variables' as *;

.btn {
  background: $primary;  // 直接使用
}

// 配置模块 — 带 !default 的变量可被覆盖
@use 'variables' with (
  $primary: #c45125,
  $radius: 12px
);

// ========== @forward 转发（forwardCode） ==========
// @forward — 转发模块（库作者必备）

// 场景：index.scss 作为统一入口

// _variables.scss
$primary: #e85d04;
$radius: 8px;

// _mixins.scss
@mixin flex-center {
  display: flex;
  align-items: center;
  justify-content: center;
}

// _functions.scss
@function rem($px) {
  @return $px / 16px * 1rem;
}

// index.scss — 统一定义公共 API
@forward 'variables';
@forward 'mixins';
@forward 'functions';

// 使用者只需要导入一个文件
@use 'foundation';

.btn {
  background: foundation.$primary;
  @include foundation.flex-center;
  padding: foundation.rem(16px);
}

// 控制转发内容 — hide / show
@forward 'variables' hide $internal-color;
@forward 'mixins' show flex-center, card;

// 转发时加前缀
@forward 'colors' as color-*;
// 使用: color.$primary → 不对，是加前缀到成员名

// ========== @import 对比（importCode） ==========
// @import — 旧方式（已弃用，但仍常见）

// 传统写法
@import 'variables';
@import 'mixins';
@import 'reset';

.btn {
  background: $primary;  // 全局污染，不知道从哪来
}

// @import 的问题：
✗ 所有变量/mixin 全局污染
✗ 多次导入可能重复输出
✗ 没有命名空间，命名冲突风险大
✗ 难以追踪依赖来源
✗ 无法配置模块参数

// 迁移建议：
1. 新项目直接用 @use / @forward
2. 旧项目逐步迁移，先从底层模块开始
3. 使用 sass-migrator 工具自动迁移
4. 保留 @import 仅用于 CSS 文件

// 对比总结
| 特性        | @import | @use |
|------------|---------|------|
| 命名空间   | ❌ 全局   | ✅ 有  |
| 可配置     | ❌      | ✅ with |
| 单次加载   | ❌ 易重复 | ✅ 模块级 |
| 私有成员   | ❌      | ✅ -前缀 |
| 推荐程度   | ⚠️ 弃用   | ✅ 推荐 |
