// ========== @debug（debugCode） ==========
// @debug — 输出变量值到控制台（开发调试）
@mixin fluid-type($min, $max) {
  $diff: $max - $min;
  @debug "差值计算: #{$diff}";
  font-size: clamp(#{$min}px, 1rem + #{$diff}vw, #{$max}px);
}

.title {
  @include fluid-type(16, 24);
}

// 控制台输出：
Debug: 差值计算: 8

// 调试复杂计算
@function calculate-gutter($cols) {
  $result: (100% / $cols) * 0.1;
  @debug "Gutter for #{$cols} cols: #{$result}";
  @return $result;
}

// ========== @warn（warnCode） ==========
// @warn — 发出警告但继续编译
@mixin spacing($value) {
  @if $value % 8 != 0 {
    @warn "间距 #{$value}px 不在 8px 网格系统上";
  }
  margin-bottom: $value + px;
}

.card {
  @include spacing(20); // 触发警告
}

// ========== @error（errorCode） ==========
// @error — 抛出错误并终止编译
@function get-color($name) {
  $colors: (
    primary: #e85d04,
    secondary: #f4a261,
    danger: #dc2626
  );
  
  @if not map-has-key($colors, $name) {
    @error "颜色 '#{$name}' 不存在。可用值：#{map-keys($colors)}";
  }
  
  @return map-get($colors, $name);
}

.btn {
  background: get-color(prinary); // 拼写错误，编译终止
}
