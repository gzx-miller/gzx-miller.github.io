// ========== @if 条件（ifCode） ==========
// @if / @else if / @else — 条件判断
@mixin button-variant($type) {
  @if $type == primary {
    background: #e85d04;
    color: #fff;
    &:hover { background: #c45125; }
  } @else if $type == secondary {
    background: #fff;
    color: #8b4513;
    border: 2px solid #e0a06a;
  } @else if $type == danger {
    background: #dc2626;
    color: #fff;
  } @else {
    @warn "未知按钮类型: #{$type}";
    background: #ccc;
  }
}

.btn-primary { @include button-variant(primary); }
.btn-secondary { @include button-variant(secondary); }
.btn-danger { @include button-variant(danger); }

// 三元表达式 — 简单条件
$is-dark: true;
.text { color: if($is-dark, #fff, #333); }

// ========== @for 循环（forCode） ==========
// @for — 循环（从数字到数字）
// through: 包含结束值 | to: 不包含结束值

// 生成间距工具类
@for $i from 1 through 6 {
  .m-#{$i} { margin: #{$i * 8}px; }
  .mt-#{$i} { margin-top: #{$i * 8}px; }
  .mb-#{$i} { margin-bottom: #{$i * 8}px; }
  .p-#{$i} { padding: #{$i * 8}px; }
}

// 编译结果
.m-1 { margin: 8px; }
.m-2 { margin: 16px; }
.m-3 { margin: 24px; }
...
.m-6 { margin: 48px; }

// 生成宽度百分比类
@for $i from 1 through 12 {
  .w-#{$i}-12 { width: percentage($i / 12); }
}

// ========== @each 遍历（eachCode） ==========
// @each — 遍历列表或 Map（最常用）

// 遍历列表
$colors: primary #e85d04, secondary #f4a261, danger #dc2626;

@each $name, $color in $colors {
  .text-#{$name} { color: $color; }
  .bg-#{$name} { background: $color; }
  .border-#{$name} { border-color: $color; }
}

// 遍历 Map（更清晰）
$breakpoints: (
  sm: 480px,
  md: 768px,
  lg: 1024px,
  xl: 1280px
);

@each $name, $width in $breakpoints {
  @media (min-width: $width) {
    .hidden-#{$name} { display: none; }
  }
}

// 多值遍历
$icons: home "\f015", user "\f007", search "\f002";
@each $name, $code in $icons {
  .icon-#{$name}:before { content: $code; }
}

// ========== @while 循环（whileCode） ==========
// @while — 条件循环（较少用）
$i: 1;
@while $i <= 6 {
  .fs-#{$i} { font-size: 12px + $i * 2; }
  $i: $i + 1;
}

// 编译结果
.fs-1 { font-size: 14px; }
.fs-2 { font-size: 16px; }
.fs-3 { font-size: 18px; }
.fs-4 { font-size: 20px; }
.fs-5 { font-size: 22px; }
.fs-6 { font-size: 24px; }

// 注意事项
• 容易写错导致无限循环
• 大多数场景 @for / @each 更清晰
• 适合需要动态步长的场景
