<script setup lang="ts">
import { ref } from 'vue'
// uni-app 的页面生命周期以组合式 API 形式从 '@dcloudio/uni-app' 导入
import { onLoad, onShow, onUnload } from '@dcloudio/uni-app'

interface Course {
  id: number
  title: string
}

const courses = ref<Course[]>([])
const keyword = ref('')

onLoad((options) => {
  // 读取路由参数并做一次性初始化
  console.log('路由参数', options?.id)
  fetchCourses()
})

onShow(() => {
  console.log('页面显示，可刷新数据')
})

onUnload(() => {
  console.log('页面销毁，清理资源')
})

function fetchCourses() {
  courses.value = [
    { id: 1, title: 'Vue3 组合式 API' },
    { id: 2, title: 'uni-app 跨端开发' },
  ]
}
</script>

<template>
  <view>
    <input v-model="keyword" placeholder="搜索课程" />
    <view v-for="c in courses" :key="c.id" class="row">{{ c.title }}</view>
  </view>
</template>

<style scoped>
.row {
  padding: 20rpx;
  border-bottom: 1rpx solid #efc48d;
}
</style>
