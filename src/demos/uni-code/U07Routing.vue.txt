<script>
export default {
  methods: {
    openCourse(id) {
      // 推入新页，保留当前页，可返回
      uni.navigateTo({ url: `/pages/course/course?id=${id}` })
    },

    finishLogin() {
      // 替换当前页，返回时不再回到登录页
      uni.redirectTo({ url: '/pages/home/home' })
    },

    switchTab() {
      // 切换底部 tab 页面
      uni.switchTab({ url: '/pages/mine/mine' })
    },

    goBack() {
      // 返回上一页，delta 可一次返回多层
      uni.navigateBack({ delta: 1 })
    },
  },
}
</script>

<template>
  <view class="nav">
    <button @tap="openCourse(1)">进入课程详情</button>
    <button @tap="goBack">返回上一页</button>
    <button @tap="switchTab">切到「我的」tab</button>
    <button @tap="finishLogin">替换为首页</button>
  </view>
</template>
