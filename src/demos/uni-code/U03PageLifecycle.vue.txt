<script>
export default {
  data() {
    return {
      courseTitle: '',
    }
  },

  onLoad(options) {
    // 页面首次创建：读取路由参数并做一次性初始化
    this.courseId = options.id
    this.fetchCourse()
  },

  onShow() {
    // 每次回到前台：刷新可能变化的数据
    this.refreshProgress()
  },

  onReady() {
    // 首次渲染完成：可访问 DOM / 初始化第三方
    console.log('页面已渲染完成')
  },

  onHide() {
    // 页面切走（未销毁）：暂停计时、节流轮询
    this.pauseCountdown()
  },

  onUnload() {
    // 页面销毁：清理定时器与监听
    this.clearTimers()
  },

  methods: {
    fetchCourse() {},
    refreshProgress() {},
    pauseCountdown() {},
    clearTimers() {},
  },
}
</script>

<template>
  <view class="course-detail">
    <text class="title">{{ courseTitle }}</text>
  </view>
</template>
