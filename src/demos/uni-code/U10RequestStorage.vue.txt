<template>
  <view class="page">
    <button type="primary" :loading="loading" @tap="fetchCourses">加载课程</button>

    <view v-if="courses.length" class="list">
      <view v-for="course in courses" :key="course.id" class="row">
        <text class="title">{{ course.title }}</text>
        <text class="tag">{{ course.tag }}</text>
      </view>
    </view>
    <text v-else class="empty">暂无课程数据</text>
  </view>
</template>

<script>
export default {
  data() {
    return {
      courses: [],
      loading: false,
    }
  },

  onLoad() {
    // 先读缓存，让页面秒开，再异步刷新
    const cached = uni.getStorageSync('courses')
    if (cached) this.courses = cached
    this.fetchCourses()
  },

  methods: {
    // 把 uni.request 封装成返回 Promise 的请求，统一处理 token 与错误
    request(options) {
      return new Promise((resolve, reject) => {
        const token = uni.getStorageSync('token')
        uni.request({
          url: 'https://api.squirrel-demo.cn' + options.url,
          method: options.method || 'GET',
          data: options.data,
          header: {
            'Content-Type': 'application/json',
            Authorization: token ? `Bearer ${token}` : '',
          },
          timeout: 10000,
          success(res) {
            if (res.statusCode === 401) {
              uni.removeStorageSync('token')
              return reject(new Error('登录已过期'))
            }
            if (res.statusCode >= 200 && res.statusCode < 300) {
              resolve(res.data)
            } else {
              reject(new Error(`请求失败：${res.statusCode}`))
            }
          },
          fail: reject,
        })
      })
    },

    async fetchCourses() {
      this.loading = true
      try {
        const list = await this.request({ url: '/courses' })
        this.courses = list
        // 数据写入本地缓存，下次进入秒开
        uni.setStorageSync('courses', list)
      } catch (e) {
        uni.showToast({ title: e.message, icon: 'none' })
      } finally {
        this.loading = false
      }
    },
  },
}
</script>
